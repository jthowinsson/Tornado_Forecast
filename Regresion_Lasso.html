import numpy as np

import numpy as np
import pandas as pd
from sklearn.metrics import mean_absolute_percentage_error, mean_squared_error, r2_score
from statsmodels.stats.diagnostic import acorr_ljungbox
from scipy.stats import jarque_bera

# Ensure grid_lasso and grid_ridge are loaded
if 'grid_lasso' not in globals() or 'grid_ridge' not in globals():
    from joblib import load
    grid_lasso = load('grid_lasso.joblib')
    grid_ridge = load('grid_ridge.joblib')

# Haciendo predicciones con el mejor modelo de Lasso
y_pred_lasso = grid_lasso.best_estimator_.predict(X_test)

# Haciendo predicciones con el mejor modelo de Ridge
y_pred_ridge = grid_ridge.best_estimator_.predict(X_test)

# Calculando las métricas para Lasso
mape_lasso = mean_absolute_percentage_error(y_test, y_pred_lasso)
rmse_lasso = np.sqrt(mean_squared_error(y_test, y_pred_lasso))
r2_lasso = r2_score(y_test, y_pred_lasso)

# Calculando Ljung-Box p-value para Lasso
lb_test_lasso = acorr_ljungbox(y_test - y_pred_lasso, lags=[10])
lb_p_value_lasso = lb_test_lasso['lb_pvalue'].iloc[0]  # Accede al valor p del primer lag (lag 10)

# Calculando Jarque-Bera p-value para Lasso
jb_p_value_lasso = jarque_bera(y_test - y_pred_lasso)[1]

# Calculando las métricas para Ridge
mape_ridge = mean_absolute_percentage_error(y_test, y_pred_ridge)
rmse_ridge = np.sqrt(mean_squared_error(y_test, y_pred_ridge))
r2_ridge = r2_score(y_test, y_pred_ridge)

# Calculando Ljung-Box p-value para Ridge
lb_test_ridge = acorr_ljungbox(y_test - y_pred_ridge, lags=[10])
lb_p_value_ridge = lb_test_ridge['lb_pvalue'].iloc[0]  # Accede al valor p del primer lag (lag 10)

# Calculando Jarque-Bera p-value para Ridge
jb_p_value_ridge = jarque_bera(y_test - y_pred_ridge)[1]

# Resultados para Lasso
metrics_lasso = {
    "MAPE": mape_lasso,
    "RMSE": rmse_lasso,
    "R Cuadrado": r2_lasso,
    "Ljung-Box p-value": lb_p_value_lasso,
    "Jarque-Bera p-value": jb_p_value_lasso
}

# Resultados para Ridge
metrics_ridge = {
    "MAPE": mape_ridge,
    "RMSE": rmse_ridge,
    "R Cuadrado": r2_ridge,
    "Ljung-Box p-value": lb_p_value_ridge,
    "Jarque-Bera p-value": jb_p_value_ridge
}

# Crear un DataFrame combinado para las métricas de Lasso y Ridge
metrics_combined_df = pd.DataFrame({
    "Lasso": metrics_lasso,
    "Ridge": metrics_ridge
}).T

# Redondear los valores a dos decimales
metrics_combined_df = metrics_combined_df.round(2)

# Mostrar la tabla combinada
print("Métricas combinadas para Lasso y Ridge:")
print(metrics_combined_df)
